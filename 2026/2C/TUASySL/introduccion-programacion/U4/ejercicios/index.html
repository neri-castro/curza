# ============================================================
# PROBLEMA 5: Calculadora de Costo de Backup
# ============================================================

COSTO_POR_GB = 12.50
DESCUENTO_DIARIO = 0.15
DESCUENTO_SEMANAL = 0.08
FRECUENCIA_MAX = 30


def es_frecuencia_valida(frecuencia_dias):
    """Retorna True si la frecuencia está entre 1 y 30 días inclusive."""
    # frecuencia_dias : int
    return 1 <= frecuencia_dias <= FRECUENCIA_MAX


def calcular_costo_backup(frecuencia_dias, gigabytes):
    """Calcula el costo mensual del backup según frecuencia y tamaño."""
    # frecuencia_dias : int
    # gigabytes       : float
    costo_base: float = gigabytes * COSTO_POR_GB
    descuento: float
    costo_final: float

    if frecuencia_dias == 1:
        descuento = DESCUENTO_DIARIO
    elif frecuencia_dias <= 7:
        descuento = DESCUENTO_SEMANAL
    else:
        descuento = 0.0

    costo_final = costo_base * (1 - descuento)
    return costo_final


# --- Programa Principal ---
if __name__ == "__main__":
    frecuencia: int = int(input("Frecuencia de backup (días): "))
    if es_frecuencia_valida(frecuencia):
        gigas: float = float(input("Tamaño en GB: "))
        costo: float = calcular_costo_backup(frecuencia, gigas)
        print(f"Costo mensual: ${costo:.2f}")
    else:
        print("Error: frecuencia fuera de rango (1-30 días)")

# ============================================================
# PROBLEMA 6: Sistema de Diagnóstico de Servidor
# ============================================================

TEMP_MAX_CPU = 80.0
USO_MAX_CPU  = 90.0
USO_MAX_RAM  = 85.0


def es_temperatura_normal(temp_cpu):
    """Retorna True si la temperatura de CPU está dentro del rango seguro."""
    # temp_cpu : float
    return temp_cpu < TEMP_MAX_CPU


def es_carga_aceptable(uso_cpu, uso_ram):
    """Retorna True si ambos usos de CPU y RAM están dentro del límite."""
    # uso_cpu : float
    # uso_ram : float
    return uso_cpu < USO_MAX_CPU and uso_ram < USO_MAX_RAM


def diagnosticar_servidor(temp_cpu, uso_cpu, uso_ram):
    """Retorna el diagnóstico del servidor: CRÍTICO, ALERTA u ÓPTIMO."""
    # temp_cpu : float
    # uso_cpu  : float
    # uso_ram  : float
    temp_normal: bool = es_temperatura_normal(temp_cpu)
    carga_ok: bool    = es_carga_aceptable(uso_cpu, uso_ram)
    estado: str

    if not temp_normal and not carga_ok:
        estado = "CRÍTICO"
    elif not temp_normal or not carga_ok:
        estado = "ALERTA"
    else:
        estado = "ÓPTIMO"

    return estado


# --- Programa Principal ---
if __name__ == "__main__":
    temp: float    = float(input("Temperatura CPU (°C): "))
    cpu: float     = float(input("Uso de CPU (%): "))
    ram: float     = float(input("Uso de RAM (%): "))
    diagnostico: str = diagnosticar_servidor(temp, cpu, ram)
    print(f"Estado del servidor: {diagnostico}")

# ============================================================
# PROBLEMA 2: Facturación de Almacenamiento con Descuento
# ============================================================

PRECIO_BASE_GB      = 150.0
DESCUENTO_LIBRE     = 0.20
DESCUENTO_EDUCATIVO = 0.10


def calcular_cuota_mensual(tipo_licencia, gigas_almacenados):
    """Calcula la cuota mensual aplicando descuento según tipo de licencia."""
    # tipo_licencia      : str
    # gigas_almacenados  : int
    costo_base: float = gigas_almacenados * PRECIO_BASE_GB
    cuota_final: float

    if tipo_licencia == 'libre':
        cuota_final = costo_base * (1 - DESCUENTO_LIBRE)
    elif tipo_licencia == 'educativa':
        cuota_final = costo_base * (1 - DESCUENTO_EDUCATIVO)
    else:
        cuota_final = costo_base

    return cuota_final


# --- Programa Principal ---

    costo: float  = calcular_cuota_mensual(licencia, gigas)
    print(f"Costo mensual: ${costo:.2f}")

def es_nota_valida(nota_evaluacion):
    """Retorna True si la nota está en el rango válido 0-10."""
    # nota_evaluacion : int
    return 0 <= nota_evaluacion <= 10


def determinar_condicion_alumno(nota_uno, nota_dos):
    """Determina la condición académica del alumno según sus dos notas."""
    # nota_uno : int
    # nota_dos : int
    condicion: str

    if nota_uno >= 7 and nota_dos >= 7:
        condicion = 'Promociona'
    elif nota_uno >= 4 and nota_dos >= 4:
        condicion = 'Aprueba'
    else:
        condicion = 'Recupera'

    return condicion

# ============================================================
# PROBLEMA 4: Control de Acceso al Centro de Cómputos
# ============================================================

HORA_INICIO_TURNO = 8
HORA_FIN_TURNO    = 20


def permitir_acceso(es_administrador, hora_ingreso):
    """Retorna True si se autoriza el acceso al centro de cómputos."""
    # es_administrador : bool
    # hora_ingreso     : int  (0-23)
    en_horario_habil: bool = HORA_INICIO_TURNO <= hora_ingreso <= HORA_FIN_TURNO
    return es_administrador or en_horario_habil


# --- Programa Principal ---
if __name__ == "__main__":
    es_admin: bool = input("¿Es administrador/a? (s/n): ").lower() == 's'
    hora: int     = int(input("Hora de ingreso (0-23): "))

    if permitir_acceso(es_admin, hora):
        print("Acceso Concedido al Centro de Cómputos")
    else:
        print("Acceso Denegado: Horario o Rol No Autorizado")
